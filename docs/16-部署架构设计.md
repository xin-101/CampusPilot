# CampusPilot 部署架构设计

## 一、部署概述

### 1.1 部署目标
CampusPilot部署架构旨在提供简单、可靠、可扩展的部署方案，支持开发、测试、生产三种环境。

### 1.2 部署原则
1. **容器化**：使用Docker容器化部署
2. **编排化**：使用Docker Compose编排服务
3. **配置外化**：配置文件与代码分离
4. **环境隔离**：不同环境使用不同配置
5. **一键部署**：提供一键部署脚本

## 二、环境规划

### 2.1 环境分类

| 环境 | 用途 | 数据库 | Agent |
|------|------|--------|-------|
| dev | 本地开发 | 本地MySQL | Mock |
| test | 测试环境 | 测试数据库 | Mock/FastGPT |
| prod | 生产环境 | 生产数据库 | FastGPT |

## 三、Docker配置

### 3.1 前端Dockerfile

```dockerfile
# campuspilot-web/Dockerfile

# 构建阶段
FROM node:20-alpine AS builder

WORKDIR /app

COPY package*.json ./
RUN npm ci

COPY . .
RUN npm run build

# 运行阶段
FROM nginx:alpine

COPY --from=builder /app/dist /usr/share/nginx/html
COPY nginx.conf /etc/nginx/conf.d/default.conf

EXPOSE 80

CMD ["nginx", "-g", "daemon off;"]
```

### 3.2 后端Dockerfile

```dockerfile
# campuspilot/Dockerfile

# 构建阶段
FROM maven:3.9-eclipse-temurin-21 AS builder

WORKDIR /app

COPY pom.xml .
RUN mvn dependency:go-offline

COPY src ./src
RUN mvn clean package -DskipTests

# 运行阶段
FROM eclipse-temurin:21-jre-alpine

WORKDIR /app

COPY --from=builder /app/target/*.jar app.jar

EXPOSE 8080

ENTRYPOINT ["java", "-jar", "app.jar"]
```

### 3.3 Nginx配置

```nginx
# campuspilot-web/nginx.conf

server {
    listen 80;
    server_name localhost;
    root /usr/share/nginx/html;
    index index.html;

    # 前端路由
    location / {
        try_files $uri $uri/ /index.html;
    }

    # API代理
    location /api/ {
        proxy_pass http://backend:8080;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
    }

    # 静态资源缓存
    location ~* \.(js|css|png|jpg|jpeg|gif|ico|svg)$ {
        expires 1y;
        add_header Cache-Control "public, immutable";
    }

    # Gzip压缩
    gzip on;
    gzip_types text/plain text/css application/json application/javascript text/xml application/xml application/xml+rss text/javascript;
}
```

```yaml
    image: mysql:8.0

      - mysql-data:/var/lib/mysql

      - campuspilot-network

      - "6379:6379"

    build:
```

### 4.2 生产环境

```yaml
# docker-compose.prod.yml

version: '3.8'

services:
  mysql:
    image: mysql:8.0
    container_name: campuspilot-mysql-prod
    ports:
      - "3306:3306"
    environment:
      MYSQL_ROOT_PASSWORD: ${DB_ROOT_PASSWORD}
      MYSQL_DATABASE: ${DB_NAME}
      MYSQL_USER: ${DB_USERNAME}
      MYSQL_PASSWORD: ${DB_PASSWORD}
    volumes:
      - mysql-data:/var/lib/mysql
      - ./database/init.sql:/docker-entrypoint-initdb.d/init.sql
    command: --default-authentication-plugin=mysql_native_password
    networks:
      - campuspilot-network
    restart: always

  redis:
    image: redis:7-alpine
    container_name: campuspilot-redis-prod
    ports:
      - "6379:6379"
    command: redis-server --appendonly yes
    volumes:
      - redis-data:/data
    networks:
      - campuspilot-network
    restart: always

  backend:
    build:
      context: ./campuspilot
      dockerfile: Dockerfile
    container_name: campuspilot-backend-prod
    ports:
      - "8080:8080"
    environment:
      - SPRING_PROFILES_ACTIVE=prod
      - DB_HOST=mysql
      - DB_PORT=3306
      - DB_NAME=${DB_NAME}
      - DB_USERNAME=${DB_USERNAME}
      - DB_PASSWORD=${DB_PASSWORD}
      - JWT_SECRET=${JWT_SECRET}
      - AGENT_PROVIDER=fastgpt
      - FASTGPT_URL=${FASTGPT_URL}
      - FASTGPT_API_KEY=${FASTGPT_API_KEY}
    depends_on:
      - mysql
      - redis
    networks:
      - campuspilot-network
    restart: always

  frontend:
    build:
      context: ./campuspilot-web
      dockerfile: Dockerfile

      - "80:80"

      - backend
    networks:
      - campuspilot-network

    image: nginx:alpine

    depends_on:
      - frontend
      - backend
    networks:
      - campuspilot-network
```

## 七、部署脚本

### 7.1 开发环境启动脚本

```bash
#!/bin/bash
# scripts/start-dev.sh

echo "启动CampusPilot开发环境..."

# 检查Docker是否运行
if ! docker info > /dev/null 2>&1; then
    echo "错误：Docker未运行，请先启动Docker"
    exit 1
fi

# 启动服务
docker-compose -f docker-compose.dev.yml up -d

# 等待服务启动
echo "等待服务启动..."
sleep 30

# 检查服务状态
docker-compose -f docker-compose.dev.yml ps

echo "开发环境启动完成！"
echo "前端访问：http://localhost"
echo "后端API：http://localhost:8080"
echo "MySQL：localhost:3306"
```

```bash
#!/bin/bash
# scripts/backup-db.sh

echo "备份CampusPilot数据库..."

# 配置
BACKUP_DIR="./backups"
DATE=$(date +%Y%m%d_%H%M%S)
BACKUP_FILE="$BACKUP_DIR/campuspilot_$DATE.sql"

# 创建备份目录
mkdir -p $BACKUP_DIR

# 备份数据库
docker exec campuspilot-mysql-prod mysqldump -u root -p$DB_ROOT_PASSWORD campuspilot > $BACKUP_FILE

find $BACKUP_DIR -name "*.gz" -mtime +30 -delete

echo "清理旧备份完成"
```

### 8.1 健康检查

```java
@RestController
@RequestMapping("/api/health")
public class HealthController {
    
    @Autowired
    private DataSource dataSource;
    
    @Autowired
    private AgentProvider agentProvider;
    
    @GetMapping
    public HealthCheckResponse health() {
        HealthCheckResponse response = new HealthCheckResponse();
        response.setStatus("UP");
        response.setTimestamp(new Date());
        
        // 检查数据库
        try {
            dataSource.getConnection();
            response.setDatabase("UP");
        } catch (Exception e) {
            response.setDatabase("DOWN");
            response.setStatus("DOWN");
        }
        
        // 检查Agent
        try {
            if (agentProvider.healthCheck()) {
                response.setAgent("UP");
            } else {
                response.setAgent("DOWN");
                response.setStatus("DOWN");
            }
        } catch (Exception e) {
            response.setAgent("DOWN");
            response.setStatus("DOWN");
        }
        
        return response;
    }
}
```
